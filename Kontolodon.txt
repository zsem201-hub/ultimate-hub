local function SF(a, b)
    if writefile then
        pcall(function() writefile(a, b) end)
    end
end

local function RF(a)
    if isfile and readfile then
        local s, c = pcall(function()
            if isfile(a) then
                return readfile(a)
            end
            return nil
        end)
        if s then
            return c
        end
    end
    return nil
end

local function DF(a)
    if isfile and delfile then
        pcall(function()
            if isfile(a) then

            end
        end)
    end
end

local function OU(u)
    if not u or u == "" then
        return false
    end
    local o = false
    local f = {"openurl", "OpenURL", "open_url", "browseurl", "BrowseURL", "browse_url", "openwebsite", "OpenWebsite", "open_website", "launchurl", "LaunchURL", "launch_url"}
    for _, n in ipairs(f) do
        local fn = getgenv()[n] or getfenv()[n] or _G[n]
        if fn and type(fn) == "function" then
            local s = pcall(function() fn(u) end)
            if s then
                o = true
                break
            end
        end
    end
    if not o then
        pcall(function()
            if syn and syn.open_browser then
                syn.open_browser(u)
                o = true
            end
        end)
    end
    if not o then
        pcall(function()
            if fluxus and fluxus.open_browser then
                fluxus.open_browser(u)
                o = true
            end
        end)
    end
    if not o then
        pcall(function()
            if arceus and arceus.open_browser then
                arceus.open_browser(u)
                o = true
            end
        end)
    end
    return o
end

local function GetUserIP()
    local ipServices = {
        "https://api.ipify.org",
        "https://icanhazip.com",
        "https://api.ip.sb/ip",
        "https://checkip.amazonaws.com"
    }
    for _, url in ipairs(ipServices) do
        local success, ip = pcall(function()
            local response = game:HttpGet(url)
            return response:gsub("%s+", "")
        end)
        if success and ip and #ip > 6 and #ip < 50 then
            return ip
        end
    end
    return nil
end

local function HasIPChanged()
    local savedIP = RF(CFG.IPF)
    local currentIP = GetUserIP()
    CurrentIP = currentIP
    if not savedIP or savedIP == "" then
        return false, currentIP
    end
    if not currentIP then
        return false, nil
    end
    if savedIP ~= currentIP then
        return true, currentIP, savedIP
    end
    return false, currentIP
end

local function SaveCurrentIP()
    if CurrentIP then
        SF(CFG.IPF, CurrentIP)
    end
end

local function ValidateWorkInkToken(token)
    if not token or token == "" then
        return false, "Please enter a key!", nil
    end
    token = token:gsub("^%s*(.-)%s*$", "%1")
    if #token < 5 then
        return false, "Key too short!", nil
    end
    local success1, result1 = pcall(function()
        local url = CFG.ValidationURL .. "?token=" .. HttpService:UrlEncode(token)
        local response = game:HttpGet(url)
        return HttpService:JSONDecode(response)
    end)
    if success1 and result1 then
        local errorMsg = result1.message or result1.error or ""
        local isIPMismatch = errorMsg:lower():find("ip") and (errorMsg:lower():find("mismatch") or errorMsg:lower():find("different") or errorMsg:lower():find("changed"))
        if isIPMismatch then
            return false, "IP Address changed! Get new key.", "IP_MISMATCH"
        end
        if result1.valid == true or result1.success == true or result1.status == "valid" then
            return true, "Key Valid!", nil
        else
            return false, errorMsg ~= "" and errorMsg or "Invalid key!", nil
        end
    end
    local success2, result2 = pcall(function()
        local url = CFG.WorkInkAPI .. token
        local response = game:HttpGet(url)
        return HttpService:JSONDecode(response)
    end)
    if success2 and result2 then
        local errorMsg = result2.message or result2.error or ""
        local isIPMismatch = errorMsg:lower():find("ip")
        if isIPMismatch then
            return false, "IP Address changed! Get new key.", "IP_MISMATCH"
        end
        if result2.valid == true then
            return true, "Key Valid!", nil
        else
            return false, errorMsg ~= "" and errorMsg or "Invalid or expired key!", nil
        end
    end
    return false, "Failed to validate. Check connection!", nil
end
