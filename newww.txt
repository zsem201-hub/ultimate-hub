local function SF(a, b) if writefile then pcall(function() writefile(a, b) end) end end
local function RF(a) if isfile and readfile then local s, c = pcall(function() if isfile(a) then return readfile(a) end return nil end) if s then return c end end return nil end
local function DF(a) if isfile and delfile then pcall(function() if isfile(a) then delfile(a) end end) end end

local function FK()
    local s, r = pcall(function()
        local res = game:HttpGet(CFG.KU)
        res = res:gsub("^%s*(.-)%s*$", "%1")
        if not res:match("^return") then res = "return " .. res end
        local fn = loadstring(res)
        if fn then return fn() end
        return nil
    end)
    if s and r then return r end
    return nil
end

local function IE(d)
    if not d then return false end
    local s, r = pcall(function()
        local y, m, dy = d:match("(%d+)-(%d+)-(%d+)")
        if y and m and dy then
            local t = os.time({year = tonumber(y), month = tonumber(m), day = tonumber(dy), hour = 23, min = 59, sec = 59})
            return os.time() > t
        end
        return false
    end)
    return s and r
end

local function VK(k)
    if not KD then return false, "Failed to fetch key data!" end
    if IE(KD.Expiry) then return false, "Key expired!" end
    if k == KD.Key then return true, KD.Message or "Key valid!" end
    return false, "Invalid key!"
end

local function SC(t) if setclipboard then pcall(function() setclipboard(t) end) end end

local function OU(u)
    if not u or u == "" then return false end
    local o = false
    local f = {"openurl", "OpenURL", "open_url", "browseurl", "BrowseURL", "browse_url", "openwebsite", "OpenWebsite", "open_website", "launchurl", "LaunchURL", "launch_url"}
    for _, n in ipairs(f) do
        local fn = getgenv()[n] or getfenv()[n] or _G[n]
        if fn and type(fn) == "function" then
            local s = pcall(function() fn(u) end)
            if s then o = true break end
        end
    end
    if not o then pcall(function() if syn and syn.open_browser then syn.open_browser(u) o = true end end) end
    if not o then pcall(function() if fluxus and fluxus.open_browser then fluxus.open_browser(u) o = true end end) end
    if not o then pcall(function() if arceus and arceus.open_browser then arceus.open_browser(u) o = true end end) end
    return o
end
